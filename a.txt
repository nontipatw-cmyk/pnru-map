        // 2. เพิ่ม Layer แผนที่จาก OpenStreetMap
        L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
            attribution: '© OpenStreetMap contributors'
        }).addTo(map);

        // 3. ข้อมูลพิกัด (GeoJSON) ที่ใส่ "type" เพื่อแยกสีแล้ว
        var geoData = {

            { "type": "Feature", "properties": { "name": "อาคารเรียน อุตสาหการ", "desc": "ที่นั่งประจำสาขา", "type": "building" }, "geometry": { "type": "Polygon", "coordinates": [[[100.588432, 13.878204], [100.588322, 13.878023], [100.588750, 13.877757], [100.588848, 13.877919], [100.588865, 13.878058], [100.588432, 13.878204]]] } },

            { "type": "Feature", "properties": { "name": "อาคาร 42 (โอ่งแดง)", "desc": "อาคารเรียนรวม", "type": "building" }, "geometry": { "type": "Polygon", "coordinates": [[[100.588323, 13.877345], [100.588208, 13.877159], [100.588574, 13.876939], [100.588323, 13.877345]]] } },

        };

        // 4. ฟังก์ชันจัดการสีและสไตล์ตามประเภท (Type)
        function getStyle(feature) {
            switch (feature.properties.type) {
                case 'gate':     return { color: "#ff4d4d", weight: 4, fillOpacity: 0.2 }; // สีแดง (ประตู)
                case 'building': return { color: "#1e90ff", weight: 2, fillOpacity: 0.5 }; // สีฟ้า (ตึก)
                case 'canteen':  return { color: "#ffa500", weight: 2, fillOpacity: 0.6 }; // สีส้ม (โรงอาหาร)
                case 'seating':  return { color: "#32cd32", weight: 2, dashArray: '5, 10', fillOpacity: 0.3 }; // สีเขียวเส้นประ (ที่นั่ง)
                default:         return { color: "#999", weight: 1 };
            }
        }

        // 5. นำข้อมูลขึ้นแผนที่
        L.geoJSON(geoData, {
            style: getStyle,
            onEachFeature: function (feature, layer) {
                // ผูก Popup แสดงชื่อและคำอธิบาย
                var popupContent = "<b>" + feature.properties.name + "</b><br>" + (feature.properties.desc || "ไม่มีข้อมูลเพิ่มเติม");
                layer.bindPopup(popupContent);
                
                // เปลี่ยนสีขอบเมื่อเอาเมาส์ไปชี้ (Optional)
                layer.on('mouseover', function () { this.setStyle({ weight: 5, color: '#666' }); });
                layer.on('mouseout', function () { L.geoJSON().resetStyle(this); layer.setStyle(getStyle(feature)); });
            }
        }).addTo(map);
